### while

```c
#include <stdio.h>
int main(void)
{
    long x = 0;
    long n = 0;

    scanf_s("%d", &x);

    while (x > 0) {
        n++;
        x /= 10;
        printf("%d %d\n", x, n);
    }
    printf("您输入的位数是:%d\n", n);
    return 0;
}
```

### do while

```c
#include <stdio.h>
int main(void)
{
    int x;
    int n = 0;

    scanf_s("%d", &x);

    do {
        x /= 10;
        n++;
    } while (x > 0);
    printf("%d", n);
    return 0;
}
```

### for

```c
for(i = 1; i <= n; n++);
```

**for 循环里有三个语句：**
- 第一个语句是 **初始条件**
- 第二个语句是 **循环继续的条件**
- 第三个语句是 **循环每轮要做的动作**

> 每一个表达式都可以省略

**循环适用条件：**
- 有固定次数 → 用 **for**
- 如果必须执行一次 → 用 **do while**
- 其他情况 → 用 **while**

## if else

> else 总是和最近的 if 匹配（除了大括号）

```c
#include<stdio.h>
int main(void)
{
    int max = 0;
    int a,b,c;
    printf("请输入三个要比较大小的数 并用空格间隔开\n");
    scanf_s("%d %d %d", &a, &b, &c);

    if (a < b) {
        if (b < c) {
            max = c;
        }else {
            max = b;
        }
    }else {
        if (a < c) {
            max = c;
        }else {
            max = a;
        }
    }
    printf("%d", max);
    return 0;
}
```

> **养成好习惯**：即使一条 `if` 也要用 `{}`

## else if

```c
#include<stdio.h>
int main()
{
    int abs, x;
    scanf_s("%d", &x);
    if (x > 0) {
        abs = 1;
    } else if (x == 0) {
        abs = 0;
    } else {
        abs = -1;
    }
    printf("%d", abs);
    return 0;
}
```

## switch case

- **使用 switch 必须是整数类型**

- **在 switch 中 break 比 case 更重要**

  - case 只是决定从哪里开始的位置

  - 而 break 决定的是程序的终止

**语法格式：**

```c
switch(type)
{
    case 1:
        语句;
        break;
    case 2:
        语句;
        break;
    // ...
    default:
        语句;
        break;
}
```

**例题：**

> 用 switch 时，如果一个 case 里有需要声明，则需要用大括号括起来

```c
switch(n)
{
    case 1:
        printf("I love Luogu!\n");
        break;
    case 2:
        printf("6 4\n");
        break;
    case 3: {
        int num = 14, p = 4;
        int x = num / p;
        printf("%d %d %d\n", x, x * p, num - x * p);
        break;
    }
    case 4: {
        float x = 500.0 / 3;
        printf("%f\n", x);
        break;
    }
    default:
        break;
}
```

## break 与 continue

- **break**：跳出循环
- **continue**：跳过本次循环中余下的语句，直接进入下一轮循环

```c
#include<stdio.h>

int main()
{
    int x, i;
    scanf("%d", &x);
    int isprime = 1;
    for (i = 2; i < x; i++) {
        if (x % i == 0) {
            isprime = 0;
            break;
        }
    }
    if (isprime == 1) {
        printf("是素数\n");
    } else {
        printf("不是素数\n");
    }
    return 0;
}
```
