# 链表

```cpp
#include <iostream>
#include <string.h>
using namespace std;
const int N = 100003;
struct Node
{
    int val;
    Node *next;
} *head;
int main()
{
    for (int i = 0; i < N; i++)
    {
        Node *p = new Node();
        p->val = i;
        p->next = head;
        head = p;
    }
    for (Node *p = head; p; p = p->next) cout << p->val << " ";
    cout << endl;
    return 0;
}
```


如果 是指针 用->访问成员

## 链表遍历

```cpp
#include<bits/stdc++.h>
using namespace std;

struct Node
{
    int val;
    Node* next;
    Node(int _val) : val(_val), next(NULL) {}
};

int main()
{
    Node* p = new Node(1);
    Node* q = new Node(2);
    Node* l = new Node(3);
    p->next = q;
    q->next = l;

    Node* head = p;
    // 遍历
    for (Node* i = head; i != NULL; i = i->next)
        cout << i->val << endl;
    return 0;
}
```

## 链表添加节点

头插法

```cpp
#include<bits/stdc++.h>
using namespace std;

struct Node
{
    int val;
    Node* next;
    Node(int _val) : val(_val), next(NULL) {}
};

int main()
{
    Node* p = new Node(1);
    Node* q = new Node(2);
    Node* l = new Node(3);
    p->next = q;
    q->next = l;
    Node* head = p;

    Node* o = new Node(0);
    o->next = head;
    head = o;
    for (Node* i = head; i; i = i->next)
        cout << i->val << endl;
    return 0;
}
```

## 链表节点删除

在遍历的时候把这个点跳过去
把要删除的点的上一个点的next指向要删除的下一个节点的地址

```cpp
int main()
{
    Node* p = new Node(1);
    Node* q = new Node(2);
    Node* l = new Node(3);
    p->next = q;
    q->next = l;
    Node* head = p;

    p->next = l;
    for (Node* i = head; i; i = i->next)
        cout << i->val << endl;
    return 0;
}
```
