# 例题

## 鸡兔同笼
```cpp
#include<stdio.h>
int main(void)
{
int x;
int y;
int rabit = 0;
int hen = 0;
printf("输入鸡兔数:");
scanf_s("%d", &x);
printf("输入脚数:");
scanf_s("%d", &y);
rabit=(y-2*x)/2;
hen = x - rabit;

if (hen<=0 || rabit<= 0) {
printf("无解");
}
else {
printf("兔子:%d\n鸡:%d\n", rabit, hen);
}
return 0;
}
```
## 移花接木
```cpp
#include <stdio.h>
int main(void)
{
float a,b,c, d, e, f;

scanf_s("%f %f", &a, &b);

c = a - (int)a;
d = b - (int)b;
e = (int)a + d;
f = (int)b + c;

printf("%.2f %.2f", e, f);
return 0;
}
```
## 阶乘

```cpp
#include <stdio.h>

int main(void)
{
    int i = 1;
    int fact = 1;
    int n;
    scanf_s("%d", &n);
    while (i <= n) {
        fact *= i;
        i++;
    }
    printf("%d!=%d", n, fact);
    return 0;
}
```

## 猜字

```cpp
#include<stdio.h>
#include<stdlib.h>
#include<time.h>
int main()
{
    srand(time(0));
    int i = 1;
    int number;
    int n = rand() % 100;
    scanf("%d", &number);
    while (n != number) {
        if (number < n) {
            printf("太小了！\n");
        } else if (number > n) {
            printf("太大了！\n");
        }
        scanf("%d", &number);
        i++;
    }
    printf("恭喜你，猜了%d次猜对了！\n", i);
    return 0;
}
```

## 倒序

```cpp
#include<stdio.h>

int main()
{
    int x;
    int y;
    int z = 0;
    scanf("%d", &x);
    while (x > 0)
    {
        y = x % 10;
        z = z * 10 + y;
        x /= 10;
    }
    printf("%d\n", z);
    return 0;
}
```

对于一个整数
%10 得到个位数
/10 除掉个位数
