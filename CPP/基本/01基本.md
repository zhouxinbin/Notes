```cpp
#include <stdio.h>

int main(void)
{
    int i = 1;
    int fact = 1;
    int n;
    scanf("%d", &n);
    while (i <= n) {
        fact *= i;
        i++;
    }
    printf("%d!=%d", n, fact);
    return 0;
}
```
